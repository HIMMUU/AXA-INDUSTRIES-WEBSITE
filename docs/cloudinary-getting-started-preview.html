Next.js / React Cloudinary SDK Snippet

Generate optimized delivery URLs with automatic format selection and quality compression:

import { cld } from '@/lib/cloudinary';
import { fill } from '@cloudinary/url-gen/actions/resize';
import { auto } from '@cloudinary/url-gen/qualifiers/format';
import { auto as qualityAuto } from '@cloudinary/url-gen/qualifiers/quality';

export function ProductImage() {
  const myImage = cld.image('samples/coffee')
    .resize(fill().width(1000).height(1000))
    .format(auto())
    .quality(qualityAuto());

  return <img src={myImage.toURL()} alt="Optimized Product" />;
}
